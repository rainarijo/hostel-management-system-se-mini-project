107def build_profile_pdf(profile):

109 profile_values, allocation_rows = profile_pdf_data(profile) 

110 lines = ["Student Profile", ""] 

111 for label, value in profile_values: 

112 lines.append(f"{label}: {value}") 

113 

114 lines.extend(["", "Room Allocation"]) 

115 if allocation_rows: 

116 for index, row in enumerate(allocation_rows, start=1): 

117 lines.append(f"Allocation {index}") 

118 labels = ( 

119 "Hostel", "Room", "Bed", "Academic year", 

120 "Start date", "End date", "Status", 

121 ) 

122 for label, value in zip(labels, row): 

123 lines.append(f"{label}: {value}") 

124 lines.append("") 

125 else: 

126 lines.append("No allocation records.") 

127 

128 # Match the existing project's dependency-free PDF strategy. 

129 page_chunks = [lines[index:index + 42] for index in range(0, len(lines), 42)] or [[]] 

130 objects = [ 

131 b"%PDF-1.4\n%\xe2\xe3\xcf\xd3\n", 

132 b"<< /Type /Catalog /Pages 2 0 R >>", 

133 b"", 

134 b"<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>", 

135 ] 

136 page_refs = [] 

137 for chunk in page_chunks: 

138 page_id = len(objects) 

139 content_id = page_id + 1 

140 page_refs.append(f"{page_id} 0 R") 

141 commands = ["BT", "/F1 9 Tf", "40 555 Td", "12 TL"] 

142 for line in chunk: 

143 commands.append(f"({_pdf_escape(line)}) Tj") 

144 commands.append("T*") 

145 commands.append("ET") 

146 content = "\n".join(commands).encode("latin-1") 

147 objects.append( 

148 f"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 842 595] " 

149 f"/Resources << /Font << /F1 3 0 R >> >> " 

150 f"/Contents {content_id} 0 R >>".encode("ascii") 

151 ) 

152 objects.append( 

153 b"<< /Length %d >>\nstream\n%s\nendstream" % 

154 (len(content), content) 

155 ) 

156 

157 objects[2] = ( 

158 f"<< /Type /Pages /Kids [{' '.join(page_refs)}] " 

159 f"/Count {len(page_refs)} >>".encode("ascii") 

160 ) 

161 

162 output = bytearray(objects[0]) 

163 offsets = [0] 

164 for object_id, body in enumerate(objects[1:], start=1): 

165 offsets.append(len(output)) 

166 output.extend(f"{object_id} 0 obj\n".encode("ascii")) 

167 output.extend(body) 

168 output.extend(b"\nendobj\n") 

169 

170 xref_offset = len(output) 

171 output.extend(f"xref\n0 {len(objects)}\n".encode("ascii")) 

172 output.extend(b"0000000000 65535 f \n") 

173 for offset in offsets[1:]: 

174 output.extend(f"{offset:010d} 00000 n \n".encode("ascii")) 

175 output.extend( 

176 f"trailer\n<< /Size {len(objects)} /Root 1 0 R >>\n" 

177 f"startxref\n{xref_offset}\n%%EOF\n".encode("ascii")
